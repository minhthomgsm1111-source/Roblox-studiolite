local function showNotification(msg, isError)
	notifyLabel.Text = msg
	if isError then
		notifyLabel.BackgroundColor3 = Color3.fromRGB(255, 70, 70)
		notifyLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
	else
		notifyLabel.BackgroundColor3 = Color3.fromRGB(255, 200, 50)
		notifyLabel.TextColor3 = Color3.fromRGB(30, 30, 30)
	end

	notifyLabel.Visible = true
	notifyLabel.TextTransparency = 0
	notifyLabel.BackgroundTransparency = 0

	task.delay(2.5, function()
		local tween = TweenService:Create(notifyLabel, TweenInfo.new(0.25), {TextTransparency = 1, BackgroundTransparency = 1})
		tween:Play()
		tween.Completed:Connect(function()
			notifyLabel.Visible = false
		end)
	end)
end

-- Trạng thái buff từ code (dùng os.clock() để đếm thời gian chính xác)
local usedCodes = {}
local doubleMoneyEndTime = 0
local buyOneGetOneEndTime = 0

local function isDoubleMoneyActive()
	return os.clock() < doubleMoneyEndTime
end

local function isBuyOneGetOneActive()
	return os.clock() < buyOneGetOneEndTime
end

local function formatMMSS(sec)
	local s = math.max(0, math.ceil(sec))
	local m = math.floor(s / 60)
	local rem = s % 60
	return string.format("%02d:%02d", m, rem)
end

-- Cập nhật đồng hồ đếm ngược hiệu lực code ngay cạnh ô Tiền
RunService.Heartbeat:Connect(function()
	local now = os.clock()
	local remX2 = doubleMoneyEndTime - now
	local remBogo = buyOneGetOneEndTime - now

	local nextX = 162

	if remX2 > 0 then
		x2TimerLabel.Visible = true
		x2TimerLabel.Position = UDim2.new(0, nextX, 1, -50)
		x2TimerLabel.Text = "x2 Tiền: " .. formatMMSS(remX2)
		nextX = nextX + 137
	else
		x2TimerLabel.Visible = false
	end

	if remBogo > 0 then
		bogoTimerLabel.Visible = true
		bogoTimerLabel.Position = UDim2.new(0, nextX, 1, -50)
		bogoTimerLabel.Text = "Mua 1 tặng 1: " .. formatMMSS(remBogo)
	else
		bogoTimerLabel.Visible = false
	end
end)

local function equipToolFromInventory(tool)
	if not tool or not tool:IsA("Tool") then return end
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum and tool.Parent then
		if tool.Parent == char then
			hum:UnequipTools()
		else
			hum:EquipTool(tool)
			showNotification("Đang cầm: " .. tool.Name, false)
		end
	end
end

local function refreshInventoryList()
	for _, child in ipairs(invScroll:GetChildren()) do
		if child:IsA("TextButton") or child:IsA("TextLabel") then
			child:Destroy()
		end
	end

	local collectedTools = {}
	local bp = player:FindFirstChild("Backpack") or backpack
	for _, item in ipairs(bp:GetChildren()) do
		if item:IsA("Tool") then
			table.insert(collectedTools, {tool = item, equipped = false})
		end
	end

	local char = player.Character
	if char then
		for _, item in ipairs(char:GetChildren()) do
			if item:IsA("Tool") then
				table.insert(collectedTools, {tool = item, equipped = true})
			end
		end
	end

	for _, entry in ipairs(collectedTools) do
		local toolRef = entry.tool
		local itemBtn = Instance.new("TextButton")
		itemBtn.Size = UDim2.new(1, 0, 0, 34)
		itemBtn.BackgroundColor3 = entry.equipped and Color3.fromRGB(0, 145, 75) or Color3.fromRGB(52, 52, 52)
		itemBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
		itemBtn.Font = Enum.Font.GothamBold
		itemBtn.TextSize = 13
		itemBtn.Text = (entry.equipped and "[Đang cầm] " or "") .. toolRef.Name
		itemBtn.Parent = invScroll

		local c = Instance.new("UICorner")
		c.CornerRadius = UDim.new(0, 6)
		c.Parent = itemBtn

		itemBtn.MouseButton1Click:Connect(function()
			equipToolFromInventory(toolRef)
			task.wait(0.05)
			if invFrame.Visible then
				refreshInventoryList()
			end
		end)
	end

	invScroll.CanvasSize = UDim2.new(0, 0, 0, #collectedTools * 40)
end

btnInv.MouseButton1Click:Connect(function()
	invFrame.Visible = not invFrame.Visible
	if invFrame.Visible then
		refreshInventoryList()
	end
end)

local function isHarvestedCropTool(item)
	return item:IsA("Tool") and item:GetAttribute("IsHarvestedCrop") == true
end

local function getCropPrices(tool)
	local savedPrice = tool:GetAttribute("SellPrice")
	local basePrice = savedPrice

	if not basePrice then
		local kg = tonumber(string.match(tool.Name, "%[([%d%.]+)kg%]")) or 1
		if string.find(tool.Name, "Cây Tre") then
			if kg < 10 then
				basePrice = math.floor(15000 + (kg * 8400))
			else
				basePrice = math.floor(100500 + ((kg - 10) * 10500))
			end
		elseif string.find(tool.Name, "Hoa Đỏ") then
			if kg < 3 then
				basePrice = math.floor(450 + (kg * 830))
			else
				basePrice = math.floor(3050 + ((kg - 3) * 1150))
			end
		elseif string.find(tool.Name, "Táo") then
			if kg < 4 then
				basePrice = math.floor(200 + (kg * 195))
			else
				basePrice = math.floor(1050 + ((kg - 4) * 320))
			end
		else
			if kg >= 10 then
				basePrice = math.floor(700 + ((kg - 10) * 60))
			else
				basePrice = math.floor(40 + (kg * 59))
			end
		end
	end

	local finalPrice = isDoubleMoneyActive() and (basePrice * 2) or basePrice
	return finalPrice, basePrice
end

-- Hàm mọc cây Carrot (10s)
local function growCarrotPlant(plantPos, plotPart)
	local weightKG, rewardMoney
	if math.random(1, 100) <= 4 then
		weightKG = math.random(1010, 1800) / 100
		rewardMoney = math.random(700, 1200)
	else
		weightKG = math.random(30, 995) / 100
		rewardMoney = math.floor(40 + (weightKG * 59))
	end

	local scale = math.clamp(0.7 + (weightKG * 0.16), 0.75, 3.8)
	local finalBodySize = Vector3.new(1.2, 2.2, 1.2) * scale
	local finalLeafSize = Vector3.new(0.8, 1.0, 0.8) * scale
	local startBodySize = Vector3.new(0.25, 0.35, 0.25)
	local startLeafSize = Vector3.new(0.2, 0.25, 0.2)
	local surfaceY = plotPart.Position.Y + (plotPart.Size.Y / 2)

	local plantModel = Instance.new("Model")
	plantModel.Name = "GrowingCrop"
	plantModel:SetAttribute("CropKind", "Carrot")
	plantModel:SetAttribute("WeightKG", weightKG)
	plantModel:SetAttribute("Scale", scale)
	plantModel:SetAttribute("RewardMoney", rewardMoney)
	plantModel.Parent = mapFolder

	local body = Instance.new("Part")
	body.Name = "Body"
	body.Size = startBodySize
	body.Position = Vector3.new(plantPos.X, surfaceY + (startBodySize.Y / 2) - 0.1, plantPos.Z)
	body.Anchored = true
	body.CanCollide = false
	body.Color = Color3.fromRGB(255, 120, 0)
	body.Material = Enum.Material.SmoothPlastic
	body.Parent = plantModel

	local leaf = Instance.new("Part")
	leaf.Name = "Leaf"
	leaf.Size = startLeafSize
	leaf.Position = Vector3.new(plantPos.X, surfaceY + startBodySize.Y + (startLeafSize.Y / 2) - 0.15, plantPos.Z)
	leaf.Anchored = true
	leaf.CanCollide = false
	leaf.Color = Color3.fromRGB(40, 200, 60)
	leaf.Material = Enum.Material.Grass
	leaf.Parent = plantModel

	local growInfo = TweenInfo.new(10, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
	TweenService:Create(body, growInfo, {
		Size = finalBodySize,
		Position = Vector3.new(plantPos.X, surfaceY + (finalBodySize.Y / 2) - 0.15, plantPos.Z)
	}):Play()
	TweenService:Create(leaf, growInfo, {
		Size = finalLeafSize,
		Position = Vector3.new(plantPos.X, surfaceY + finalBodySize.Y + (finalLeafSize.Y / 2) - 0.2, plantPos.Z)
	}):Play()

	task.delay(10, function()

-- Hàm mọc Cây Tre (30s): dài ít nhất 4m, vừa to vừa dài theo kg, dưới 10kg < 100k$, trên 10kg > 100k$
local function growBambooPlant(plantPos, plotPart)
	local weightKG, rewardMoney
	if math.random(1, 100) <= 32 then
		-- Trên 10kg -> trên 100,000 $ (càng nhiều gam/kg càng nhiều tiền)
		weightKG = math.random(1005, 2200) / 100
		rewardMoney = math.floor(100500 + ((weightKG - 10) * 10500))
	else
		-- Dưới 10kg -> dưới 100,000 $ (càng nhiều gam/kg càng nhiều tiền)
		weightKG = math.random(120, 995) / 100
		rewardMoney = math.floor(15000 + (weightKG * 8400))
	end

	-- Dài ít nhất 4m (1m ~ 3.5 studs -> ít nhất 14 studs), dài và to tỷ lệ thuận với kg
	local heightMeters = 4.0 + (weightKG * 0.48) -- ~4.5m đến ~14.5m
	local finalHeightStuds = heightMeters * 3.5
	local thicknessScale = math.clamp(0.85 + (weightKG * 0.14), 0.95, 4.0)
	local finalDiameter = 0.85 * thicknessScale

	local surfaceY = plotPart.Position.Y + (plotPart.Size.Y / 2)

	local plantModel = Instance.new("Model")
	plantModel.Name = "GrowingCrop"
	plantModel:SetAttribute("CropKind", "Bamboo")
	plantModel:SetAttribute("WeightKG", weightKG)
	plantModel:SetAttribute("Scale", thicknessScale)
	plantModel:SetAttribute("RewardMoney", rewardMoney)
	plantModel.Parent = mapFolder

	-- Thân tre chính (Cylinder xoay dọc)
	local body = Instance.new("Part")
	body.Name = "Body"
	body.Shape = Enum.PartType.Cylinder
	body.Size = Vector3.new(0.8, 0.25, 0.25)
	body.CFrame = CFrame.new(plantPos.X, surfaceY + 0.4, plantPos.Z) * CFrame.Angles(0, 0, math.rad(90))
	body.Anchored = true
	body.CanCollide = false
	body.Color = Color3.fromRGB(70, 185, 55)
	body.Material = Enum.Material.SmoothPlastic
	body.Parent = plantModel

	local growDuration = 30
	local growInfo = TweenInfo.new(growDuration, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

	TweenService:Create(body, growInfo, {
		Size = Vector3.new(finalHeightStuds, finalDiameter, finalDiameter),
		CFrame = CFrame.new(plantPos.X, surfaceY + (finalHeightStuds / 2), plantPos.Z) * CFrame.Angles(0, 0, math.rad(90))
	}):Play()

	-- Các đốt tre (5 đốt dọc thân cây) mọc lớn cùng thân
	local nodeCount = 5
	for n = 1, nodeCount do
		local ratio = n / (nodeCount + 1)
		local nodeRing = Instance.new("Part")
		nodeRing.Name = "BambooNode_" .. n
		nodeRing.Shape = Enum.PartType.Cylinder
		nodeRing.Size = Vector3.new(0.1, 0.3, 0.3)
		nodeRing.CFrame = CFrame.new(plantPos.X, surfaceY + (0.8 * ratio), plantPos.Z) * CFrame.Angles(0, 0, math.rad(90))
		nodeRing.Anchored = true
		nodeRing.CanCollide = false
		nodeRing.Color = Color3.fromRGB(45, 130, 35)
		nodeRing.Material = Enum.Material.SmoothPlastic
		nodeRing.Parent = plantModel

		local targetRingThick = finalDiameter * 1.16
		local targetRingHeight = math.clamp(0.24 * thicknessScale, 0.22, 0.65)
		TweenService:Create(nodeRing, growInfo, {
			Size = Vector3.new(targetRingHeight, targetRingThick, targetRingThick),
			CFrame = CFrame.new(plantPos.X, surfaceY + (finalHeightStuds * ratio), plantPos.Z) * CFrame.Angles(0, 0, math.rad(90))
		}):Play()
	end

	-- Tán lá tre ở các đốt trên
	for l = 1, 3 do
		local leafRatio = 0.72 + (l * 0.08)
		local sideSign = (l % 2 == 0) and 1 or -1
		local leaf = Instance.new("Part")
		leaf.Name = "BambooLeaf_" .. l
		leaf.Size = Vector3.new(0.3, 0.08, 0.15)
		leaf.CFrame = CFrame.new(plantPos.X, surfaceY + 0.6, plantPos.Z)
		leaf.Anchored = true
		leaf.CanCollide = false
		leaf.Color = Color3.fromRGB(50, 205, 65)
		leaf.Material = Enum.Material.Grass
		leaf.Parent = plantModel

		local finalLeafSize = Vector3.new(2.2 * thicknessScale, 0.14, 0.75 * thicknessScale)
		local leafOffsetX = sideSign * ((finalDiameter / 2) + (finalLeafSize.X * 0.42))
		TweenService:Create(leaf, growInfo, {
			Size = finalLeafSize,
			CFrame = CFrame.new(plantPos.X + leafOffsetX, surfaceY + (finalHeightStuds * leafRatio), plantPos.Z) * CFrame.Angles(0, 0, math.rad(sideSign * 18))
		}):Play()
	end

	task.delay(growDuration, function()
		if not plantModel or not plantModel.Parent then return end
		plantModel:SetAttribute("ReadyToHarvest", true)

		local harvestPrompt = Instance.new("ProximityPrompt")
		harvestPrompt.ObjectText = string.format("Cây Tre (%.2fkg - %.1fm)", weightKG, heightMeters)
		harvestPrompt.ActionText = "Thu hoạch"
		harvestPrompt.HoldDuration = 0
		harvestPrompt.MaxActivationDistance = 15
		harvestPrompt.RequiresLineOfSight = false
		harvestPrompt.Parent = body

		local harvested = false
		harvestPrompt.Triggered:Connect(function(plr)
			if plr == player and not harvested then
				harvested = true
				plantModel:Destroy()
				local bp = player:FindFirstChild("Backpack") or backpack
				local bambooTool = createHarvestedBambooTool(weightKG, thicknessScale, rewardMoney)
				bambooTool.Parent = bp
				showNotification(string.format("Thu hoạch: Cây Tre %.2fkg (dài %.1fm)", weightKG, heightMeters), false)
				if invFrame.Visible then refreshInventoryList() end
			end
		end)
	end)
end

-- Hàm mọc Cây Táo: 81s lớn cây, mỗi 90s mọc trái trên cây
local function growAppleTree(plantPos, plotPart)
	local treeWeightKG
	if math.random(1, 100) <= 30 then
		treeWeightKG = math.random(2010, 3200) / 100
	else
		treeWeightKG = math.random(600, 1990) / 100
	end

	local treeHeightMeters = math.clamp(4 + ((treeWeightKG - 6) * 0.65), 4, 21)
	local totalHeightStuds = treeHeightMeters * 3.5

	local trunkHeight = totalHeightStuds * 0.65
	local trunkWidth = math.clamp(1.2 + (treeWeightKG * 0.11), 1.4, 5.2)
	local canopySize = Vector3.new(totalHeightStuds * 0.62, totalHeightStuds * 0.45, totalHeightStuds * 0.62)

	local surfaceY = plotPart.Position.Y + (plotPart.Size.Y / 2)

	local treeModel = Instance.new("Model")
	treeModel.Name = "GrowingCrop"
	treeModel:SetAttribute("CropKind", "AppleTree")
	treeModel.Parent = mapFolder

	local trunk = Instance.new("Part")
	trunk.Name = "Body"
	trunk.Size = Vector3.new(0.4, 1.0, 0.4)
	trunk.Position = Vector3.new(plantPos.X, surfaceY + 0.5, plantPos.Z)
	trunk.Anchored = true
	trunk.CanCollide = true
	trunk.Color = Color3.fromRGB(105, 65, 35)
	trunk.Material = Enum.Material.Wood
	trunk.Parent = treeModel

	local canopy = Instance.new("Part")
	canopy.Name = "Canopy"
	canopy.Shape = Enum.PartType.Ball
	canopy.Size = Vector3.new(1.2, 1.2, 1.2)
	canopy.Position = Vector3.new(plantPos.X, surfaceY + 1.2, plantPos.Z)
	canopy.Anchored = true
	canopy.CanCollide = false
	canopy.Color = Color3.fromRGB(45, 165, 55)
	canopy.Material = Enum.Material.Grass
	canopy.Parent = treeModel

	local canopyTop = Instance.new("Part")
	canopyTop.Name = "CanopyTop"
	canopyTop.Shape = Enum.PartType.Ball
	canopyTop.Size = Vector3.new(0.9, 0.9, 0.9)
	canopyTop.Position = Vector3.new(plantPos.X, surfaceY + 1.5, plantPos.Z)
	canopyTop.Anchored = true
	canopyTop.CanCollide = false
	canopyTop.Color = Color3.fromRGB(55, 185, 65)
	canopyTop.Material = Enum.Material.Grass
	canopyTop.Parent = treeModel

	local finalTrunkPos = Vector3.new(plantPos.X, surfaceY + (trunkHeight / 2), plantPos.Z)
	local finalCanopyPos = Vector3.new(plantPos.X, surfaceY + trunkHeight + (canopySize.Y * 0.25), plantPos.Z)
	local finalCanopyTopSize = canopySize * 0.72
	local finalCanopyTopPos = finalCanopyPos + Vector3.new(0, canopySize.Y * 0.34, 0)

	local growDuration = 81
	local growInfo = TweenInfo.new(growDuration, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

	TweenService:Create(trunk, growInfo, {
		Size = Vector3.new(trunkWidth, trunkHeight, trunkWidth),
		Position = finalTrunkPos
	}):Play()

	TweenService:Create(canopy, growInfo, {
		Size = canopySize,
		Position = finalCanopyPos
	}):Play()

	TweenService:Create(canopyTop, growInfo, {
		Size = finalCanopyTopSize,
		Position = finalCanopyTopPos
	}):Play()

	local fruitsFolder = Instance.new("Folder")
	fruitsFolder.Name = "AppleFruits"
	fruitsFolder.Parent = treeModel

	local function spawnAppleBatch()
		if not treeModel or not treeModel.Parent then return end

		local targetCount
		if treeWeightKG >= 20 then
			targetCount = math.random(8, 16)
		else
			targetCount = math.random(5, 8)
		end

		local currentFruits = #fruitsFolder:GetChildren()
		local toSpawn = math.max(0, targetCount - currentFruits)
		if toSpawn == 0 then return end

		local rx = canopySize.X * 0.38
		local ry = canopySize.Y * 0.32
		local rz = canopySize.Z * 0.38

		for i = 1, toSpawn do
			local appleWeightKG, appleReward
			if math.random(1, 100) <= 28 then
				appleWeightKG = math.random(405, 780) / 100
				appleReward = math.floor(1050 + ((appleWeightKG - 4) * 320))
			else
				appleWeightKG = math.random(40, 395) / 100
				appleReward = math.floor(200 + (appleWeightKG * 195))
			end

			local appleScale = math.clamp(0.75 + (appleWeightKG * 0.28), 0.85, 2.9)
			local appleRadius = 1.15 * appleScale

			local angle = (i / toSpawn) * math.pi * 2 + (math.random() * 0.8)
			local distFactor = 0.55 + (math.random() * 0.4)
			local offsetX = math.cos(angle) * rx * distFactor
			local offsetZ = math.sin(angle) * rz * distFactor
			local offsetY = -ry * (0.2 + math.random() * 0.65)

			local applePart = Instance.new("Part")
			applePart.Name = "AppleFruit"
			applePart.Shape = Enum.PartType.Ball
			applePart.Size = Vector3.new(0.2, 0.2, 0.2)
			applePart.Position = finalCanopyPos + Vector3.new(offsetX, offsetY, offsetZ)
			applePart.Anchored = true
			applePart.CanCollide = false
			applePart.Color = Color3.fromRGB(230, 35, 45)
			applePart.Material = Enum.Material.SmoothPlastic
			applePart:SetAttribute("WeightKG", appleWeightKG)
			applePart:SetAttribute("Scale", appleScale)
			applePart:SetAttribute("RewardMoney", appleReward)
			applePart.Parent = fruitsFolder

			TweenService:Create(applePart, TweenInfo.new(1.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
				Size = Vector3.new(appleRadius, appleRadius, appleRadius)
			}):Play()

			local prompt = Instance.new("ProximityPrompt")
			prompt.ObjectText = string.format("Trái Táo (%.2fkg)", appleWeightKG)
			prompt.ActionText = "Hái táo"
			prompt.HoldDuration = 0
			prompt.MaxActivationDistance = 16
			prompt.RequiresLineOfSight = false
			prompt.Parent = applePart

			local picked = false
			prompt.Triggered:Connect(function(plr)
				if plr == player and not picked then
					picked = true
					applePart:Destroy()
					local bp = player:FindFirstChild("Backpack") or backpack
					local appleTool = createHarvestedAppleTool(appleWeightKG, appleScale, appleReward)
					appleTool.Parent = bp
					showNotification(string.format("Đã hái: Táo %.2fkg", appleWeightKG), false)
					if invFrame.Visible then refreshInventoryList() end
				end
			end)
		end
	end

	task.spawn(function()
		task.wait(growDuration)
		if not treeModel or not treeModel.Parent then return end

		showNotification(string.format("Cây táo (%.1fkg - cao %.1fm) đã trưởng thành và ra trái!", treeWeightKG, treeHeightMeters), false)
		spawnAppleBatch()

		while treeModel and treeModel.Parent do
			task.wait(90)
			if not treeModel or not treeModel.Parent then break end
			spawnAppleBatch()
		end
	end)
end

-- Thu hoạch nhanh tất cả nông sản/trái cây chín của người chơi
harvestAllFruitsAction = function()
	local bp = player:FindFirstChild("Backpack") or backpack
	local harvestedCount = 0

	for _, obj in ipairs(mapFolder:GetChildren()) do
		if obj:IsA("Model") and obj.Name == "GrowingCrop" then
			local kind = obj:GetAttribute("CropKind")
			if kind == "Carrot" and obj:GetAttribute("ReadyToHarvest") == true then
				local w = obj:GetAttribute("WeightKG") or 1
				local sc = obj:GetAttribute("Scale") or 1
				local rew = obj:GetAttribute("RewardMoney") or 100
				obj:Destroy()
				createHarvestedCarrotTool(w, sc, rew).Parent = bp
				harvestedCount = harvestedCount + 1
			elseif kind == "RedFlower" and obj:GetAttribute("ReadyToHarvest") == true then
				local w = obj:GetAttribute("WeightKG") or 1
				local sc = obj:GetAttribute("Scale") or 1
				local rew = obj:GetAttribute("RewardMoney") or 1000
				obj:Destroy()
				createHarvestedFlowerTool(w, sc, rew).Parent = bp
				harvestedCount = harvestedCount + 1
			elseif kind == "Bamboo" and obj:GetAttribute("ReadyToHarvest") == true then
				local w = obj:GetAttribute("WeightKG") or 5
				local sc = obj:GetAttribute("Scale") or 1
				local rew = obj:GetAttribute("RewardMoney") or 50000
				obj:Destroy()
				createHarvestedBambooTool(w, sc, rew).Parent = bp
				harvestedCount = harvestedCount + 1
			elseif kind == "AppleTree" then
				local fruitsFolder = obj:FindFirstChild("AppleFruits")
				if fruitsFolder then
					for _, fruit in ipairs(fruitsFolder:GetChildren()) do
						if fruit:IsA("BasePart") and fruit.Name == "AppleFruit" then
							local w = fruit:GetAttribute("WeightKG") or 1
							local sc = fruit:GetAttribute("Scale") or 1
							local rew = fruit:GetAttribute("RewardMoney") or 400
							fruit:Destroy()
							createHarvestedAppleTool(w, sc, rew).Parent = bp
							harvestedCount = harvestedCount + 1
						end
					end
				end
			end
		end
	end

	if harvestedCount > 0 then
		showNotification(string.format("Đã thu hoạch %d món vào balo!", harvestedCount), false)
		if invFrame.Visible then refreshInventoryList() end
	else
		showNotification("Chưa có trái cây/nông sản nào sẵn sàng để thu hoạch!", true)
	end
end

-- 12. Cộng dồn Stack Hạt giống & Trồng trên khu đất đã mở rộng
local function addSeedToBackpack(seedName, seedColor, seedMaterial, cropType, amountToAdd)
	local amount = amountToAdd or 1
	local bp = player:FindFirstChild("Backpack") or backpack
	local char = player.Character

	local existingTool = nil
	for _, item in ipairs(bp:GetChildren()) do
		if item:IsA("Tool") and item:GetAttribute("SeedCropType") == cropType then
			existingTool = item
			break
		end
	end
	if not existingTool and char then
		local held = char:FindFirstChildOfClass("Tool")
		if held and held:GetAttribute("SeedCropType") == cropType then
			existingTool = held
		end
	end

	if existingTool then
		local currentCount = tonumber(string.match(existingTool.Name, "x(%d+)")) or 1
		local newCount = currentCount + amount
		existingTool.Name = string.format("%s (x%d)", seedName, newCount)
		existingTool.ToolTip = string.format("Số lượng: %d", newCount)
	else
		local tool = Instance.new("Tool")
		tool.Name = string.format("%s (x%d)", seedName, amount)
		tool.ToolTip = string.format("Số lượng: %d", amount)
		tool:SetAttribute("SeedCropType", cropType)
		tool:SetAttribute("BaseSeedName", seedName)

		local handle = Instance.new("Part")
		handle.Name = "Handle"
		handle.Size = Vector3.new(1.2, 1.2, 1.2)
		handle.Shape = Enum.PartType.Ball
		handle.Color = seedColor
		handle.Material = seedMaterial or Enum.Material.SmoothPlastic
		handle.Parent = tool

		tool.Parent = bp
	end

	if invFrame.Visible then
		refreshInventoryList()
	end
end

local lastPlantTick = 0

local function tryPlantAtPosition(hitPosition, targetPart)
	local now = os.clock()
	if now - lastPlantTick < 0.01 then return end

	local char = player.Character
	local heldTool = char and char:FindFirstChildOfClass("Tool")
	local cropType = heldTool and heldTool:GetAttribute("SeedCropType")
	if not heldTool or not cropType or not hitPosition or not targetPart then return end

	local selectedPlot = nil
	for _, plotData in ipairs(farmPlots) do
		if targetPart == plotData.Part then
			selectedPlot = plotData
			break
		end
	end

	if not selectedPlot then return end

	if selectedPlot.Owner ~= player.Name then
		showNotification("Đây không phải khu đất của bạn!", true)
		return
	end

	local p = selectedPlot.Part
	local clampedX = math.clamp(hitPosition.X, p.Position.X - (p.Size.X / 2) + 2.0, p.Position.X + (p.Size.X / 2) - 2.0)
	local clampedZ = math.clamp(hitPosition.Z, p.Position.Z - (p.Size.Z / 2) + 2.0, p.Position.Z + (p.Size.Z / 2) - 2.0)
	local surfaceY = p.Position.Y + (p.Size.Y / 2)
	local finalPlantPos = Vector3.new(clampedX, surfaceY, clampedZ)

	for _, obj in ipairs(mapFolder:GetChildren()) do
		if obj.Name == "GrowingCrop" then
			local bPart = obj:FindFirstChild("Body")
			if bPart then
				local dist = (Vector3.new(finalPlantPos.X, 0, finalPlantPos.Z) - Vector3.new(bPart.Position.X, 0, bPart.Position.Z)).Magnitude
				if dist < 2.2 then
					showNotification("Không thể trồng trùng lên cây khác!", true)
					return
				end
			end
		end
	end

	lastPlantTick = now

	local baseSeedName = heldTool:GetAttribute("BaseSeedName") or "Hạt giống"
	local currentCount = tonumber(string.match(heldTool.Name, "x(%d+)")) or 1
	if currentCount > 1 then
		local newCount = currentCount - 1
		heldTool.Name = string.format("%s (x%d)", baseSeedName, newCount)
		heldTool.ToolTip = string.format("Số lượng: %d", newCount)
	else
		heldTool:Destroy()
	end

	if cropType == "RedFlower" then
		growRedFlowerPlant(finalPlantPos, p)
	elseif cropType == "AppleTree" then
		growAppleTree(finalPlantPos, p)
		showNotification("Đã trồng Hạt táo! Cây sẽ lớn sau 1 phút 21 giây", false)
	elseif cropType == "Bamboo" then
		growBambooPlant(finalPlantPos, p)
		showNotification("Đã trồng Hạt tre! Cây tre sẽ mọc xong sau 30 giây", false)
	else
		growCarrotPlant(finalPlantPos, p)
	end

	if invFrame.Visible then
		refreshInventoryList()
	end
end

local function createSeedOnPath(spawnPos, seedConfig)
	local targetFolder = (mapFolder and mapFolder.Parent) and mapFolder or Workspace
	local seedPart = Instance.new("Part")
	seedPart.Name = "Handle"
	seedPart.Size = Vector3.new(1.4, 1.4, 1.4)
	seedPart.Shape = Enum.PartType.Ball
	seedPart.Position = spawnPos
	seedPart.Anchored = true
	seedPart.CanCollide = false
	seedPart.Color = seedConfig.Color
	seedPart.Material = seedConfig.Material or Enum.Material.SmoothPlastic
	seedPart.Parent = targetFolder

	local prompt = Instance.new("ProximityPrompt")
	prompt.ObjectText = seedConfig.ObjectText
	prompt.ActionText = seedConfig.Price > 0 and ("Mua (" .. seedConfig.Price .. " $)") or "Mua"
	prompt.HoldDuration = 0.2
	prompt.MaxActivationDistance = 7
	prompt.RequiresLineOfSight = false
	prompt.Parent = seedPart

	local billboard = Instance.new("BillboardGui")
	billboard.Size = UDim2.new(0, 125, 0, 42)
	billboard.StudsOffset = Vector3.new(0, 2, 0)
	billboard.AlwaysOnTop = true
	billboard.Parent = seedPart

	local label = Instance.new("TextLabel")
	label.Size = UDim2.new(1, 0, 1, 0)
	label.BackgroundTransparency = 1
	label.Text = seedConfig.LabelText
	label.TextColor3 = Color3.fromRGB(255, 255, 255)
	label.TextStrokeTransparency = 0
	label.Font = Enum.Font.GothamBold
	label.TextScaled = true
	label.Parent = billboard

	local bought = false
	prompt.Triggered:Connect(function(plr)
		if plr == player and not bought then
			if seedConfig.Price > 0 and moneyStat.Value < seedConfig.Price then
				showNotification(string.format("Không đủ tiền! Cần %d $ để mua %s", seedConfig.Price, seedConfig.ObjectText), true)
				return
			end

			bought = true
			if seedConfig.Price > 0 then
				moneyStat.Value = moneyStat.Value - seedConfig.Price
			end

			seedPart:Destroy()
			local bonusActive = isBuyOneGetOneActive()
			local qty = bonusActive and 2 or 1
			addSeedToBackpack(seedConfig.SeedName, seedConfig.Color, seedConfig.Material, seedConfig.CropType, qty)

			if bonusActive then
				showNotification("Đã mua: " .. seedConfig.ObjectText .. " (+1 TẶNG 1)", false)
			else
				showNotification("Đã mua: " .. seedConfig.ObjectText, false)
			end
		end
	end)

	return seedPart
end

local startPos = Vector3.new(centerPos.X, centerPos.Y + 1.5, caveZ + 4)
local endZ = centerPos.Z + (mapSize / 2) - 2

task.spawn(function()
	local lockedType = nil
	local otherCountAfterLock = 0
	local lastSpawnedType = nil
	local currentStreak = 0

	while player.Parent do
		task.wait(2.2)

		if not mapFolder or not mapFolder.Parent then
			mapFolder = Instance.new("Folder")
			mapFolder.Name = "CarrotFarmMap"
			mapFolder.Parent = Workspace
		end

		local allowedTypes = {}
		for idx = 1, #seedTypes do
			local blockedByLock = (lockedType == idx and otherCountAfterLock < 2)
			local blockedByStreak = (lastSpawnedType == idx and currentStreak >= 3)
			if not blockedByLock and not blockedByStreak then
				table.insert(allowedTypes, idx)
			end
		end

		if #allowedTypes == 0 then
			lockedType = nil
			otherCountAfterLock = 0
			currentStreak = 0
			for idx = 1, #seedTypes do
				if idx ~= lastSpawnedType then
					table.insert(allowedTypes, idx)
				end
			end
		end

		local pool = {}
		for _, idx in ipairs(allowedTypes) do
			local cfg = seedTypes[idx]
			if cfg.SpawnChance >= 100 or math.random(1, 100) <= cfg.SpawnChance then
				local w = cfg.SpawnWeight or 2
				for _ = 1, w do
					table.insert(pool, idx)
				end
			end
		end

		local chosenType = (#pool > 0) and pool[math.random(1, #pool)] or allowedTypes[1]
		local chosenConfig = seedTypes[chosenType]

		if chosenType == lastSpawnedType then
			currentStreak = currentStreak + 1
		else
			lastSpawnedType = chosenType
			currentStreak = 1
		end

		if lockedType ~= nil and chosenType ~= lockedType then
			otherCountAfterLock = otherCountAfterLock + 1
			if otherCountAfterLock >= 2 then
				lockedType = nil
				otherCountAfterLock = 0
			end
		end

		if currentStreak >= 3 then
			lockedType = chosenType
			otherCountAfterLock = 0
		end

		local seedPart = createSeedOnPath(startPos, chosenConfig)

		local conn
		conn = RunService.RenderStepped:Connect(function(dt)
			if not seedPart or not seedPart.Parent then
				if conn then conn:Disconnect() end
				return
			end

			seedPart.Position = seedPart.Position + Vector3.new(0, 0, SEED_SPEED * dt)

			if seedPart.Position.Z >= endZ then
				seedPart:Destroy()
				if conn then conn:Disconnect() end
			end
		end)
	end
end)
